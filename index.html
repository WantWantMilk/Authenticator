function getUserPreferences(username) { const data = loadData(); const user = data.users[username]; if (!user) return getDefaultPreferences(); let changed = false; if (!user.preferences) { user.preferences = getDefaultPreferences(); changed = true; } else { const defaults = getDefaultPreferences(); Object.keys(defaults).forEach(k => { if (user.preferences[k] === undefined) { user.preferences[k] = defaults[k]; changed = true; } }); } if (changed) saveData(data); return user.preferences; }
